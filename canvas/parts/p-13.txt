        <BarChart
          horizontal
          height={640}
          yMin={0}
          yMax={5}
          categories={CATS.map((c) => c.name)}
          series={[{ name: `${t.short} kategori puanı`, data: CATS.map((c) => Number(catScore(c, tool).toFixed(2))), tone: "info" }]}

      </Stack>
    </Grid>
  );
